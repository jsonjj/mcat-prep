#!/bin/sh
set -eu
test "$#" -eq 1 || {
  echo "usage: commands/validate-one.txt <candidate-lesson-directory>" >&2
  exit 2
}
COMMAND_DIR=${0%/*}
KIT_ROOT=$(CDPATH= cd -- "$COMMAND_DIR/.." && pwd)
REPOSITORY_ROOT=${REPOSITORY_ROOT:-$(git -C "$PWD" rev-parse --show-toplevel)}
exec node "$KIT_ROOT/commands/kit-command.mjs" validate-one "$1" --repository-root "$REPOSITORY_ROOT"
